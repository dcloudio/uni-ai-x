	let phoneSpeedTimer = 0
	let phoneSpeedPoll = 0
	let phoneSpeedRound = 0
	function phoneSpeedRun() {
		phoneSpeedRound += 1
		uniAi.llm = { provider: 'deepseek', model: 'deepseek-flash', thinking: false, webSearch: false }
		uniAi.addChat(false)
		phoneSpeedTimer = setTimeout(() => {
			const startedAt = Date.now()
			let firstBody = -1
			let firstThinking = -1
			let processingSeen = false
			console.log('[PhoneSpeed6Run20261116] start', JSON.stringify({ round: phoneSpeedRound, startedAt: startedAt, model: uniAi.llm.model }))
			phoneSpeedPoll = setInterval(() => {
				const chat = uniAi.currentChat
				if (chat == null) return
				if (chat.state == 'processing') processingSeen = true
				const msg = chat.msgList.find((item: MsgItem): boolean => item.from_uid == 'uni-ai')
				const elapsed = Date.now() - startedAt
				if (msg != null) {
					if (firstThinking < 0 && (msg.thinkContent ?? '').length > 0) {
						firstThinking = elapsed
						console.log('[PhoneSpeed6Run20261116] first-thinking', phoneSpeedRound, elapsed)
					}
					if (firstBody < 0 && msg.body.length > 0) {
						firstBody = elapsed
						console.log('[PhoneSpeed6Run20261116] first-body', phoneSpeedRound, elapsed)
					}
				}
				if ((processingSeen && chat.state != 'processing') || elapsed > 90000) {
					clearInterval(phoneSpeedPoll)
					phoneSpeedPoll = 0
					console.log('[PhoneSpeed6Run20261116] complete', JSON.stringify({ round: phoneSpeedRound, firstBodyMs: firstBody, firstThinkingMs: firstThinking, totalMs: elapsed, state: chat.state, bodyChars: msg?.body.length ?? 0, exactMatch: msg?.body == phoneSpeedExpected, rendered: msg?.rendered ?? false, error: msg?.error_msg ?? '' }))
					if (msg != null) {
						for (let offset = 0; offset < msg.body.length; offset += 500) {
							console.log('[PhoneSpeed6Run20261116] body-part', JSON.stringify({ round: phoneSpeedRound, offset: offset, text: msg.body.slice(offset, offset + 500) }))
						}
					}
					if (phoneSpeedRound < 6 && chat.state == 'completed') phoneSpeedTimer = setTimeout(phoneSpeedRun, 2000)
				}
			}, 50)
			uniAi.sendMsg(phoneSpeedPrompt).catch((error) => { console.error('[PhoneSpeed6Run20261116] send-error', error) })
		}, 500)
	}
	onMounted(() => { phoneSpeedTimer = setTimeout(phoneSpeedRun, 8000) })
	onUnmounted(() => {
		if (phoneSpeedTimer != 0) clearTimeout(phoneSpeedTimer)
		if (phoneSpeedPoll != 0) clearInterval(phoneSpeedPoll)
	})
